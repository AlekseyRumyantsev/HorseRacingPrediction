window.MegaCreator = class {
    constructor(seed) {
        this.IntToDoubleMultiplier = 1.0 / (2147483647.0 + 1.0);
        this.UIntToDoubleMultiplier = 1.0 / (4294967295.0 + 1.0);
        this.N = 624; this.M = 397;
        this.VectorA = 0x9908b0dfn; this.UpperMask = 0x80000000n; this.LowerMask = 0x7fffffffn;
        this.seed = BigInt(Math.abs(Number(seed))) & 0xffffffffn;
        this.mt = new BigInt64Array(this.N); this.mti = 0n;
        this.ResetCreator();
    }
    ResetCreator() {
        this.mt[0] = this.seed & 0xffffffffn; this.mti = 1n;
        while (this.mti < BigInt(this.N)) {
            let idx = Number(this.mti); let prevIdx = Number(this.mti - 1n);
            this.mt[idx] = (1812433253n * (this.mt[prevIdx] ^ (this.mt[prevIdx] >> 30n)) + this.mti) & 0xffffffffn;
            this.mti++;
        }
    }
    CreateNUInts() {
        let kk = 0; let y = 0n; let mag01 = [0n, this.VectorA];
        while (kk < this.N - this.M) {
            y = (this.mt[kk] & this.UpperMask) | (this.mt[kk + 1] & this.LowerMask);
            this.mt[kk] = this.mt[kk + this.M] ^ (y >> 1n) ^ mag01[Number(y & 0x1n)]; kk++;
        }
        while (kk < this.N - 1) {
            y = (this.mt[kk] & this.UpperMask) | (this.mt[kk + 1] & this.LowerMask);
            this.mt[kk] = this.mt[kk + (this.M - this.N)] ^ (y >> 1n) ^ mag01[Number(y & 0x1n)]; kk++;
        }
        y = (this.mt[this.N - 1] & this.UpperMask) | (this.mt[0] & this.LowerMask);
        this.mt[this.N - 1] = this.mt[this.M - 1] ^ (y >> 1n) ^ mag01[Number(y & 0x1n)]; this.mti = 0n;
    }
    ExtractUint() {
        if (this.mti >= BigInt(this.N)) this.CreateNUInts();
        let _y = this.mt[Number(this.mti)]; this.mti++;
        _y = _y ^ (_y >> 11n); _y = _y ^ ((_y << 7n) & 0x9d2c5680n);
        _y = _y ^ ((_y << 15n) & 0xefc60000n); return (_y ^ (_y >> 18n)) & 0xffffffffn;
    }
};

window.SuperCreator = class {
    constructor(seed) {
        this.IntToDoubleMultiplier = 1.0 / (2147483647.0 + 1.0);
        this.UIntToDoubleMultiplier = 1.0 / (4294967295.0 + 1.0);
        this.shortLag = 418; this.longLag = 1279;
        this.seed = BigInt(Math.abs(Number(seed))) & 0xffffffffn;
        this.x = new BigInt64Array(this.longLag); this.i = 0;
        this.ResetCreator();
    }
    ResetCreator() {
        let gen = new window.MegaCreator(this.seed);
        for (let k = 0; k < this.longLag; k++) this.x[k] = gen.ExtractUint();
        this.i = this.longLag;
    }
    Fill() {
        for (let k = 0; k < this.shortLag; k++) this.x[k] = (this.x[k] + this.x[k + (this.longLag - this.shortLag)]) & 0xffffffffn;
        for (let k = this.shortLag; k < this.longLag; k++) this.x[k] = (this.x[k] + this.x[k - this.shortLag]) & 0xffffffffn;
        this.i = 0;
    }
    Extract(minValue, maxValue) {
        if (minValue > maxValue) return 2147483647;
        if (this.i >= this.longLag) this.Fill();
        let _x = this.x[this.i++]; let range = maxValue - minValue;
        if (range < 0) {
            return Math.floor(minValue + (Number(_x) * this.UIntToDoubleMultiplier * (maxValue - minValue)));
        } else {
            return Math.floor(minValue + (Number(_x >> 1n) * this.IntToDoubleMultiplier * range));
        }
    }
};
